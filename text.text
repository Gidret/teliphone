if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405);
    exit;
}

// Защита: покупать могут только авторизованные пользователи
if (!isset($_SESSION['user_id'])) {
    $_SESSION['message'] = 'Чтобы совершить покупку, необходимо авторизоваться!';
    header('Location: ../authPage.php?stat=err');
    exit;
}

$phone_model = trim($_POST['phone_model'] ?? '');
$price       = trim($_POST['price'] ?? '');

if (empty($phone_model) || empty($price)) {
    $_SESSION['message'] = 'Не удалось определить параметры товара.';
    header('Location: ../home.php?stat=err');
    exit;
}

require_once '../db/conn.php';

// Фиксируем покупку в таблице purchases
$stmt = $conn->prepare('INSERT INTO purchases (user_id, phone_model, price) VALUES (:user_id, :phone_model, :price)');
$success = $stmt->execute([
    ':user_id'     => $_SESSION['user_id'],
    ':phone_model' => $phone_model,
    ':price'       => $price
]);

if ($success) {
    $_SESSION['message'] = 'Поздравляем с покупкой! Данные сохранены в вашем профиле.';
    header('Location: ../home.php?stat=ok');
} else {
    $_SESSION['message'] = 'Произошла ошибка при оформлении покупки.';
    header('Location: ../home.php?stat=err');
}
exit;
